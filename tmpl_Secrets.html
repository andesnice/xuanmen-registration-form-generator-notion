// ═══════════════════════════════════════════════════════════ // {{EVENT_NAME}} — Secrets.gs（機密元件，純後端） // Notion API token 與報名 DB ID 集中在這裡，絕不注入任何前端頁面。 // 取值一律走 notionToken_() / notionDbId_()，勿在 Code/Config 直接寫死。 // // ── 混合式存放 ────────────────────────────────────────────── // 預設用下方由產生器內嵌的常數（開箱即用，雙擊自動部署即可）。 // 要更嚴格（機密不落地在程式碼）：在 GAS 編輯器手動執行一次 // setupSecrets() 把值搬進 Script Properties，再把 NOTION_TOKEN_FALLBACK // 與 NOTION_DB_ID_FALLBACK 清成 '' 後重新部署即可。 // 之後 getter 會優先讀 Script Properties，讀不到才回退常數。 // ═══════════════════════════════════════════════════════════ // 產生器內嵌值（fallback）。設定 Script Properties 後可清空。 var NOTION_TOKEN_FALLBACK = {{NOTION_TOKEN_JS}}; var NOTION_DB_ID_FALLBACK = {{NOTION_DB_ID_JS}}; var NOTION_VERSION = '2022-06-28'; var SP_TOKEN_KEY = 'NOTION_TOKEN'; var SP_DBID_KEY = 'NOTION_DB_ID'; // 優先 Script Properties，其次內嵌常數。前端永遠拿不到這兩個值。 function notionToken_() { var v = String(PropertiesService.getScriptProperties().getProperty(SP_TOKEN_KEY) || '').trim(); return v || String(NOTION_TOKEN_FALLBACK || '').trim(); } function notionDbId_() { var v = String(PropertiesService.getScriptProperties().getProperty(SP_DBID_KEY) || '').trim(); return v || String(NOTION_DB_ID_FALLBACK || '').trim(); } // 在 GAS 編輯器手動執行一次：把內嵌常數搬進 Script Properties。 // 之後即可把上方兩個 *_FALLBACK 清成 '' 再重新部署，機密就不再留在程式碼裡。 function setupSecrets() { var props = PropertiesService.getScriptProperties(); var token = String(NOTION_TOKEN_FALLBACK || '').trim(); var dbId = String(NOTION_DB_ID_FALLBACK || '').trim(); if (token) props.setProperty(SP_TOKEN_KEY, token); if (dbId) props.setProperty(SP_DBID_KEY, dbId); var ok = !!notionToken_() && !!notionDbId_(); return ok ? '已寫入 Script Properties。之後可把 Secrets.gs 兩個 FALLBACK 常數清空再重新部署，機密即不再留在程式碼裡。' : '寫入失敗：內嵌常數為空，請先填入 token 與 DB ID，或直接改用 Script Properties。'; } // 快速自檢（可在編輯器執行）：印出目前是否取得到 token 與 DB ID，但不外洩內容。 function checkSecrets() { return { hasToken: !!notionToken_(), hasDbId: !!notionDbId_() }; }
